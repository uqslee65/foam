=== roots.ts ===

Import paths (use these exactly):
  import type { URI } from '@foam/core';
  import { RootKind, RootGroup, makeNodeId, isValidNodePath, type NodeId } from '@foam/graph-view/protocol';
  import { relativeTo, isWithinPath } from '@foam/core';

Exports:

  interface ClassifiedRoot { uri: URI; kind: RootKind; }
  interface RootClassification { roots: ClassifiedRoot[]; rootGroups: RootGroup[]; }

  classifyRoots(workspaceRoots: URI[], classification?: Map<URI, RootKind>): RootClassification
    - If classification provided, use it. Otherwise: first folder = 'vault', rest = 'sources'.
    - rootGroups: vault = { root: 'vault', label: 'Vault', color: '#4a9eff' }
                  sources = { root: 'sources', label: 'Sources', color: '#ff9f43' }
    - Pure function, no vscode import.

  resolveRoot(uri: URI, classifiedRoots: ClassifiedRoot[]): ClassifiedRoot | undefined
    - First ClassifiedRoot where isWithinPath(uri, root.uri) is true.
    - Order matters (first match wins).

  makeResourceNodeId(uri: URI, classifiedRoots: ClassifiedRoot[]): NodeId | undefined
    - Call resolveRoot. If no root, return undefined.
    - relativePath = relativeTo(uri.path, root.uri.path)
    - Validate with isValidNodePath(relativePath). If invalid, return undefined.
    - Return makeNodeId(root.kind, relativePath).

=== graph-data.ts ===

Import paths (use these exactly):
  import type { URI } from '@foam/core';
  import type { Resource, Connection } from '@foam/core';
  import { buildGraphData } from '@foam/core';
  import type { GraphData, NodeId, RootGroup } from '@foam/graph-view/protocol';
  import { classifyRoots, makeResourceNodeId } from './roots';

Exports:

  buildRootQualifiedGraphData(resources: Resource[], connections: Connection[], workspaceRoots: URI[]): { graphData: GraphData; idToUri: Map<NodeId, URI> }
    - Call classifyRoots(workspaceRoots).
    - Call buildGraphData(resources, connections, { resourceToId: uri => makeResourceNodeId(uri, classification.roots), includePlaceholders: false }).
    - Build idToUri map by iterating resources, calling makeResourceNodeId for each.
    - Return { graphData: { ...builtGraphData, rootGroups: classification.rootGroups }, idToUri }.

Key: includePlaceholders: false (behaviour change vs current graph-webview which uses true).
Key: resourceToId returns undefined for resources outside roots — excluded.
Key: never pass invalid paths to makeNodeId (pre-validates with isValidNodePath).
No vscode import.
