// @ts-expect-error The graph-view package publishes its runtime protocol as ESM while this package type-checks as CommonJS.
import { isNodeId } from '@foam/graph-view/protocol';
import type { NodeId } from '@foam/graph-view/protocol' with { 'resolution-mode': 'import' };
import type { FocusRequest } from '@foam/graph-view/protocol' with { 'resolution-mode': 'import' };

export type FocusFilePayload =
  | { ok: true; request: FocusRequest }
  | { ok: false; reason: FocusParseError };

export type FocusParseError =
  | 'invalid_shape'
  | 'wrong_workspace'
  | 'invalid_node_id'
  | 'missing_request_id';

function hasOwn(value: Record<string, unknown>, key: string): boolean {
  return Object.prototype.hasOwnProperty.call(value, key);
}

export function parseFocusFilePayload(
  input: unknown,
  expected: { workspaceId: string }
): FocusFilePayload {
  if (typeof input !== 'object' || input === null) {
    return { ok: false, reason: 'invalid_shape' };
  }

  const payload = input as Record<string, unknown>;
  if (
    typeof payload.workspace !== 'string' ||
    typeof payload.requestId !== 'string' ||
    typeof payload.nodeId !== 'string'
  ) {
    return { ok: false, reason: 'invalid_shape' };
  }

  if (payload.workspace !== expected.workspaceId) {
    return { ok: false, reason: 'wrong_workspace' };
  }

  if (payload.requestId.length === 0) {
    return { ok: false, reason: 'missing_request_id' };
  }

  if (!isNodeId(payload.nodeId)) {
    return { ok: false, reason: 'invalid_node_id' };
  }

  if (hasOwn(payload, 'center') && typeof payload.center !== 'boolean') {
    return { ok: false, reason: 'invalid_shape' };
  }

  const request: FocusRequest = {
    requestId: payload.requestId,
    nodeId: payload.nodeId as NodeId,
  };
  if (hasOwn(payload, 'center')) {
    request.center = payload.center as boolean;
  }

  return { ok: true, request };
}

export class FocusQueue {
  private readonly maxSeen: number;
  private queue: FocusRequest[] = [];
  private ready = false;
  private seen: string[] = [];

  constructor(options?: { maxSeen?: number }) {
    const maxSeen = options?.maxSeen ?? 1000;
    this.maxSeen =
      typeof maxSeen === 'number' && Number.isFinite(maxSeen)
        ? Math.max(1, Math.floor(maxSeen))
        : 1000;
  }

  enqueue(request: FocusRequest): void {
    if (
      this.seen.includes(request.requestId) ||
      this.queue.some(item => item.requestId === request.requestId)
    ) {
      return;
    }

    if (this.ready) {
      this.queue.push(request);
    } else {
      this.queue = [request];
    }
  }

  markReady(): void {
    this.ready = true;
  }

  markNotReady(): void {
    this.ready = false;
    if (this.queue.length > 1) {
      this.queue = [this.queue[this.queue.length - 1]];
    }
  }

  drain(): FocusRequest[] {
    const drained = this.queue;
    this.queue = [];
    for (const request of drained) {
      this.markSeen(request.requestId);
    }
    return drained;
  }

  markSeen(requestId: string): void {
    if (this.seen.includes(requestId)) {
      return;
    }

    this.seen.push(requestId);
    if (this.seen.length > this.maxSeen) {
      this.seen = this.seen.slice(Math.floor(this.seen.length / 2));
    }
  }

  get pending(): number {
    return this.queue.length;
  }
}

export interface FocusFs {
  readFile(path: string): Promise<string | null>;
  watchDir(
    dir: string,
    onChange: () => void,
    onError: (err: unknown) => void
  ): { close(): void };
}

export interface WatchFocusFileOptions {
  dir: string;
  filename: string;
  fs: FocusFs;
  workspaceId: string;
  onRequest: (request: FocusRequest) => void;
}

const focusWatchReattachDelayMs = 1000;

export function watchFocusFile(
  options: WatchFocusFileOptions
): { dispose(): void } {
  const filePath = `${options.dir}/${options.filename}`;
  const seen = new Set<string>();
  let watcher: { close(): void } | undefined;
  let reattachTimer: ReturnType<typeof setTimeout> | undefined;
  let disposed = false;
  let reading = false;

  async function readRequest(startup: boolean): Promise<void> {
    const content = await options.fs.readFile(filePath);
    if (content === null) {
      return;
    }

    let parsed: unknown;
    try {
      parsed = JSON.parse(content);
    } catch {
      return;
    }

    const result = parseFocusFilePayload(parsed, {
      workspaceId: options.workspaceId,
    });
    if (!result.ok || disposed) {
      return;
    }

    const { request } = result;
    if (startup) {
      seen.add(request.requestId);
      return;
    }

    if (seen.has(request.requestId)) {
      return;
    }

    seen.add(request.requestId);
    options.onRequest(request);
  }

  function readOnChange(): void {
    if (disposed || reading) {
      return;
    }

    reading = true;
    void readRequest(false)
      .catch(() => undefined)
      .finally(() => {
        reading = false;
      });
  }

  function scheduleReattach(): void {
    if (disposed || reattachTimer !== undefined) {
      return;
    }

    const timer = setTimeout(() => {
      reattachTimer = undefined;
      attachWatcher();
    }, focusWatchReattachDelayMs);
    reattachTimer = timer;

    // Unref Node timers when available; the VS Code layer owns watcher disposal.
    const unref = (timer as unknown as { unref?: () => void }).unref;
    unref?.call(timer);
  }

  function handleWatcherError(_error: unknown): void {
    if (disposed) {
      return;
    }

    watcher?.close();
    watcher = undefined;
    scheduleReattach();
  }

  function attachWatcher(): void {
    if (disposed) {
      return;
    }

    try {
      watcher = options.fs.watchDir(
        options.dir,
        readOnChange,
        handleWatcherError
      );
    } catch (error) {
      handleWatcherError(error);
    }
  }

  attachWatcher();

  reading = true;
  void readRequest(true)
    .catch(() => undefined)
    .finally(() => {
      reading = false;
    });

  return {
    dispose(): void {
      disposed = true;
      watcher?.close();
      watcher = undefined;
      if (reattachTimer !== undefined) {
        clearTimeout(reattachTimer);
        reattachTimer = undefined;
      }
    },
  };
}

import type { DenyReason, PrivacyPolicy } from './privacy';

export type OpenNodeDeny =
  | { allowed: false; reason: 'invalid_id' }
  | { allowed: false; reason: 'unknown_id'; nodeId: NodeId }
  | { allowed: false; reason: 'stale_mapping'; nodeId: NodeId; uri: URI }
  | {
      allowed: false;
      reason: 'privacy_denied';
      nodeId: NodeId;
      uri: URI;
      denyReason: DenyReason;

export type OpenNodeAllow = { allowed: true; nodeId: NodeId; uri: URI };

export type OpenNodeResult = OpenNodeAllow | OpenNodeDeny;

export function resolveNodeOpen(
  payload: unknown,
  idToUri: Map<NodeId, URI>,
  policy: PrivacyPolicy,
  isInWorkspace: (uri: URI) => boolean
): OpenNodeResult {
  if (!isNodeId(payload)) {
    return { allowed: false, reason: 'invalid_id' };
  }

  const uri = idToUri.get(payload);
  if (uri === undefined) {
    return { allowed: false, reason: 'unknown_id', nodeId: payload };
  }

  if (!isInWorkspace(uri)) {
    return { allowed: false, reason: 'stale_mapping', nodeId: payload, uri };
  }

  if (uri.scheme !== 'file') {
    return {
      allowed: false,
      reason: 'privacy_denied',
      nodeId: payload,
      uri,
      denyReason: 'outside_roots',
    };
  }

  const decision = policy.decide(uri.toFsPath());
  if (!decision.allowed) {

function isRootKind(value: unknown): value is RootKind {
  return value === 'vault' || value === 'sources';
}

function isValidNodePath(path: string): boolean {
  if (path.length === 0 || path.length > 1024 || path.startsWith('/')) {
    return false;
  }

  if (path.includes('\\') || path.includes('://')) {
    return false;
  }

  if (/^[A-Za-z]+:/.test(path) || /%2e|%2f|%5c/i.test(path)) {
    return false;
  }

  for (const character of path) {
    const codePoint = character.codePointAt(0);
    if (codePoint !== undefined && (codePoint < 0x20 || codePoint === 0x7f)) {
      return false;
    }
  }

  return path
    .split('/')
    .every((segment) => segment !== '' && segment !== '.' && segment !== '..');
}

export function parseNodeId(value: unknown): { root: RootKind; path: string } | null {
  if (typeof value !== 'string' || value.length > 1100) {
    return null;
  }

  const separator = value.indexOf(':');
  if (separator <= 0) {
    return null;
  }

  const root = value.slice(0, separator);
  const path = value.slice(separator + 1);
  if (!isRootKind(root) || !isValidNodePath(path)) {
    return null;
  }

  return { root, path };
}

export function isNodeId(value: unknown): value is NodeId {
  return parseNodeId(value) !== null;
}

export function makeNodeId(root: RootKind, path: string): NodeId {
  const value = `${root}:${path}`;
  if (!isNodeId(value)) {
    throw new RangeError('Invalid node id');
  }
  return value;
}

          .toString(),
    });

      roots: workspaceFolderFsPaths,

              type: 'didUpdateStyle',
              payload: style,

            break;

          case 'webviewDidSelectNode': {
            const result = resolveNodeOpen(
              message.payload,
              this.idToUri,

            );

export function viewConfigToStyle(config: GraphViewConfig): GraphStyle {
  const nodeColors = config.show
    ? Object.fromEntries(
        Object.entries(config.show)
          .filter(([, cfg]) => cfg.color)
          .map(([type, cfg]) => [type, cfg.color!])
      )
    : undefined;

  const showNodesOfType = config.show
    ? Object.fromEntries(
        Object.entries(config.show).map(([type, cfg]) => [
          type,
          cfg.enabled ?? true,
        ])
      )
    : undefined;

  const styleProps: Record<string, unknown> = {};
  if (config.background !== undefined)
    styleProps.background = config.background;
  if (config.fontSize !== undefined) styleProps.fontSize = config.fontSize;
  if (config.fontFamily !== undefined)
    styleProps.fontFamily = config.fontFamily;
  if (config.lineColor !== undefined) styleProps.lineColor = config.lineColor;
  if (nodeColors && Object.keys(nodeColors).length > 0)
    styleProps.node = nodeColors;

  return {
    ...(config.colorBy !== undefined ? { colorMode: config.colorBy } : {}),
    ...(config.groups !== undefined ? { groups: config.groups } : {}),
    ...(showNodesOfType !== undefined ? { showNodesOfType } : {}),
    ...(Object.keys(styleProps).length > 0 ? { style: styleProps } : {}),
  };
}

    groups: patch.groups ?? base.groups,
    showNodesOfType: { ...base.showNodesOfType, ...patch.showNodesOfType },
    style: { ...base.style, ...patch.style },
  };
}

export function resolveViewStyle(args?: ShowGraphArgs): {
  style: GraphStyle;
  view?: string;
} {
  const views: GraphViewConfig[] =
    vscode.workspace.getConfiguration('pa-hub.graph').get('views') ?? [];

  let style: GraphStyle = getGraphStyle();

  // If no view name given and no inline config, fall back to the "Default" named view
  const namedView = args?.view ?? (args?.config ? undefined : 'Default');

  if (namedView) {
    const view = views.find(v => v.name === namedView);
    if (view) {
      style = mergeStyles(style, viewConfigToStyle(view));
    }
  }

  if (args?.config) {
    style = mergeStyles(style, viewConfigToStyle(args.config));
  }

  return { style, view: args?.view };
}

export function handleActiveResourceChange(
  webviewHost: GraphWebviewHost | undefined,
  foam: Foam,
  uri: URI | undefined,
  idToUri: Map<NodeId, URI> = new Map()
) {
  if (webviewHost && uri && uri.scheme !== 'untitled') {
    const note = foam.workspace.find(uri);
    if (isSome(note)) {
      let nodeId: NodeId | undefined;
      for (const [id, resourceUri] of idToUri) {
        if (resourceUri.isEqual(uri)) {
          nodeId = id;
          break;
        }
      }

      if (!nodeId) {
        return;
      }

      webviewHost.webview.postMessage({
        type: 'didSelectNote',
        payload: nodeId,
      });
    }
  }
}

export function getNodeNavigationCommand(
  uriPath: string,
  navigateToPreview: boolean
): string {
  if (navigateToPreview && uriPath.endsWith('.md')) {
    return 'markdown.showPreview';
  }
  return 'vscode.open';
}

export function viewConfigToStyle(config: GraphViewConfig): GraphStyle {
  const nodeColors = config.show
    ? Object.fromEntries(
        Object.entries(config.show)
          .filter(([, cfg]) => cfg.color)
          .map(([type, cfg]) => [type, cfg.color!])
      )
    : undefined;

  const showNodesOfType = config.show
    ? Object.fromEntries(
        Object.entries(config.show).map(([type, cfg]) => [
          type,
          cfg.enabled ?? true,
        ])
      )
    : undefined;

  const styleProps: Record<string, unknown> = {};
  if (config.background !== undefined)
    styleProps.background = config.background;
  if (config.fontSize !== undefined) styleProps.fontSize = config.fontSize;
  if (config.fontFamily !== undefined)
    styleProps.fontFamily = config.fontFamily;
  if (config.lineColor !== undefined) styleProps.lineColor = config.lineColor;
  if (nodeColors && Object.keys(nodeColors).length > 0)
    styleProps.node = nodeColors;

  return {
    ...(config.colorBy !== undefined ? { colorMode: config.colorBy } : {}),
    ...(config.groups !== undefined ? { groups: config.groups } : {}),
    ...(showNodesOfType !== undefined ? { showNodesOfType } : {}),
    ...(Object.keys(styleProps).length > 0 ? { style: styleProps } : {}),
  };
}

export function mergeStyles(base: GraphStyle, patch: GraphStyle): GraphStyle {
  return {
    colorMode: patch.colorMode ?? base.colorMode,
    groups: patch.groups ?? base.groups,
    showNodesOfType: { ...base.showNodesOfType, ...patch.showNodesOfType },
    style: { ...base.style, ...patch.style },
  };
}

export function resolveViewStyle(args?: ShowGraphArgs): {
  style: GraphStyle;
  view?: string;
} {
  const views: GraphViewConfig[] =
    vscode.workspace.getConfiguration('pa-hub.graph').get('views') ?? [];

  let style: GraphStyle = getGraphStyle();

  // If no view name given and no inline config, fall back to the "Default" named view
  const namedView = args?.view ?? (args?.config ? undefined : 'Default');

  if (namedView) {
    const view = views.find(v => v.name === namedView);
    if (view) {
      style = mergeStyles(style, viewConfigToStyle(view));
    }
  }

  if (args?.config) {
    style = mergeStyles(style, viewConfigToStyle(args.config));
  }

  return { style, view: args?.view };
}
