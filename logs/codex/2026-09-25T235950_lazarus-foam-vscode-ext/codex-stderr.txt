=== webview-security.ts ===

Import: only node:crypto (for randomBytes). No vscode import. No other runtime deps.

Exports:

  generateNonce(): string
    - Returns a 32-byte random hex string via crypto.randomBytes(32).toString('hex').

  buildCsp(params: { cspSource: string; nonce: string }): string
    - Returns this exact CSP string (one line, semicolons separating directives):
      "default-src 'none'; script-src 'nonce-${nonce}'; style-src ${cspSource} 'unsafe-inline'; img-src ${cspSource}; font-src ${cspSource}"
    - The 'unsafe-inline' in style-src is required because force-graph 1.51.4 injects a runtime style element.

  fillGraphHtml(template: string, params: { csp: string; nonce: string; resolveAsset: (filename: string) => string }): string
    - Replaces __PA_HUB_CSP__ with params.csp (exactly 1 occurrence in the template).
    - Replaces __PA_HUB_NONCE__ with params.nonce (exactly 2 occurrences in the template: one on link, one on script).
    - Replaces data-replace attributes: pattern /data-replace (src|href)="[^"]+"/g
      For each match: extract the filename from the quoted value, call resolveAsset(filename), replace the whole match with just the attribute (src or href) = resolved URI.
      Example: data-replace src="./main.js" becomes src="https://webview-uri/main.js"
    - Returns the filled HTML string.

The template is static/dataviz/index.html (N09 output). Read it before implementing.
It has: a CSP meta with content="__PA_HUB_CSP__", a link with data-replace href="./main.css" and nonce="__PA_HUB_NONCE__", and a script with data-replace src="./main.js" and nonce="__PA_HUB_NONCE__". Two nonce placeholders, one CSP placeholder.

async function getWebviewContent(
  context: vscode.ExtensionContext,
  panel: vscode.WebviewPanel
) {
  const datavizUri = vscode.Uri.joinPath(
    context.extensionUri,
    'static',
    'dataviz'
  );
  const getWebviewUri = (fileName: string) =>
    panel.webview.asWebviewUri(vscode.Uri.joinPath(datavizUri, fileName));

  const indexHtml = new TextDecoder('utf-8').decode(
    await vscode.workspace.fs.readFile(
      vscode.Uri.joinPath(datavizUri, 'index.html')
    )
  );

  // Replace the script paths with the appropriate webview URI.
  const filled = indexHtml.replace(
    /data-replace (src|href)="[^"]+"/g,
    match => {
      const i = match.indexOf(' ');
      const j = match.indexOf('=');
      const uri = getWebviewUri(match.slice(j + 2, -1).trim());
      return match.slice(i + 1, j) + '="' + uri.toString() + '"';
    }
  );

  return filled;
}

function createPolicy(isDenied: (input: string) => boolean): PrivacyPolicy {
  const isAllowed = (input: unknown): boolean =>
    typeof input === 'string' && !isDenied(input);

  return {
    version: 'test-privacy-v1',
    decide: input => {
      const stringInput = typeof input === 'string' ? input : null;
      const allowed = isAllowed(input);
      return {
        allowed,
        reason: allowed
          ? 'allowed'
          : stringInput === null
          ? 'invalid_input'
          : 'configured_exclusion',
        input: stringInput,
        canonicalPath: null,
        root: null,
        relativePath: null,
      };
    },
    isAllowed,
    purge: <K extends string>(keys: Iterable<K>): K[] =>
      [...keys].filter(key => !isAllowed(key)),
  };
}

function createMatcher(matchedUris: URI[]): {
  matcher: IMatcher;
  refreshCount: () => number;
} {
  let refreshes = 0;
  const matches = (uri: URI) =>
    matchedUris.some(matchedUri => matchedUri.isEqual(uri));

  return {
    matcher: {
      include: ['**/*'],
      exclude: [],
      isMatch: matches,
      match: files => files.filter(matches),
      refresh: async () => {
        refreshes += 1;
      },
    },
    refreshCount: () => refreshes,
  };
}

function createDataStore(
  files: URI[],
  content: Map<string, string>
): {
  dataStore: IDataStore;
  reads: URI[];
  writes: Array<{ uri: URI; content: string }>;
  deletes: URI[];
  moves: Array<{ from: URI; to: URI }>;
  exists: URI[];
  patterns: Array<string | undefined>;
} {
  const reads: URI[] = [];
  const writes: Array<{ uri: URI; content: string }> = [];
  const deletes: URI[] = [];
  const moves: Array<{ from: URI; to: URI }> = [];
  const exists: URI[] = [];
  const patterns: Array<string | undefined> = [];

  return {
    dataStore: {
      list: async pattern => {
        patterns.push(pattern);
        return files;
      },
      read: async uri => {
        reads.push(uri);
        return content.get(uri.toString()) ?? null;
      },
      write: async (uri, value) => {
        writes.push({ uri, content: value });
      },
      delete: async uri => {
        deletes.push(uri);
      },
      move: async (from, to) => {
        moves.push({ from, to });
      },
      exists: async uri => {
        exists.push(uri);
        return content.has(uri.toString());
      },
    },
    reads,
    writes,
    deletes,
    moves,
    exists,
    patterns,
  };
}

+export function generateNonce(): string {
+  return randomBytes(32).toString('hex');

+
+export function buildCsp(params: { cspSource: string; nonce: string }): string {

+}

+  template: string,
+  params: {
+    csp: string;
+    nonce: string;
+    resolveAsset: (filename: string) => string;
+  }

+  const withSecurity = template
+    .replace('__PA_HUB_CSP__', params.csp)
+    .replace(/__PA_HUB_NONCE__/g, params.nonce);
+

+    match => {
+      const [, attribute, filename] =
+        /^data-replace (src|href)="([^"]+)"$/.exec(match)!;
+      return `${attribute}="${params.resolveAsset(filename)}"`;
+    }
+  );

+import { randomBytes } from 'node:crypto';
+
+export function generateNonce(): string {
+  return randomBytes(32).toString('hex');
+}
+
+export function buildCsp(params: { cspSource: string; nonce: string }): string {
+  return `default-src 'none'; script-src 'nonce-${params.nonce}'; style-src ${params.cspSource} 'unsafe-inline'; img-src ${params.cspSource}; font-src ${params.cspSource}`;
+}
+
+export function fillGraphHtml(
+  template: string,
+  params: {
+    csp: string;
+    nonce: string;
+    resolveAsset: (filename: string) => string;
+  }
+): string {
+  const withSecurity = template
+    .replace('__PA_HUB_CSP__', params.csp)
+    .replace(/__PA_HUB_NONCE__/g, params.nonce);
+
+  return withSecurity.replace(
+    /data-replace (src|href)="[^"]+"/g,
+    match => {
+      const [, attribute, filename] =
+        /^data-replace (src|href)="([^"]+)"$/.exec(match)!;
+      return `${attribute}="${params.resolveAsset(filename)}"`;
+    }
+  );
+}
