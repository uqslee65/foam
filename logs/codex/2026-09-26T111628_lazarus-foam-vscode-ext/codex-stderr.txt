import type { URI } from '@foam/core';
// @ts-expect-error The graph-view package publishes its runtime protocol as ESM while this package type-checks as CommonJS.
import { isNodeId } from '@foam/graph-view/protocol';
import type { NodeId } from '@foam/graph-view/protocol' with {
  'resolution-mode': 'import',
};
import type { DenyReason, PrivacyPolicy } from './privacy';

export type OpenNodeDeny =
  | { allowed: false; reason: 'invalid_id' }
  | { allowed: false; reason: 'unknown_id'; nodeId: NodeId }
  | { allowed: false; reason: 'stale_mapping'; nodeId: NodeId; uri: URI }
  | {
      allowed: false;
      reason: 'privacy_denied';
      nodeId: NodeId;
      uri: URI;
      denyReason: DenyReason;
    };

export type OpenNodeAllow = { allowed: true; nodeId: NodeId; uri: URI };

export type OpenNodeResult = OpenNodeAllow | OpenNodeDeny;

export function resolveNodeOpen(
  payload: unknown,
  idToUri: Map<NodeId, URI>,
  policy: PrivacyPolicy,
  isInWorkspace: (uri: URI) => boolean
): OpenNodeResult {
  if (!isNodeId(payload)) {
    return { allowed: false, reason: 'invalid_id' };
  }

  const uri = idToUri.get(payload);
  if (uri === undefined) {
    return { allowed: false, reason: 'unknown_id', nodeId: payload };
  }

  if (!isInWorkspace(uri)) {
    return { allowed: false, reason: 'stale_mapping', nodeId: payload, uri };
  }

  if (uri.scheme !== 'file') {
    return {
      allowed: false,
      reason: 'privacy_denied',
      nodeId: payload,
      uri,

function resolveCanonicalPath(normalisedPath: string): string | null {
  try {
    if (fs.existsSync(normalisedPath)) {
      return fs.realpathSync.native(normalisedPath);
    }

    let cursor = normalisedPath;
    const remainder: string[] = [];
    while (!fs.existsSync(cursor)) {
      const parent = path.dirname(cursor);
      if (parent === cursor) {
        return null;
      }
      remainder.unshift(path.basename(cursor));
      cursor = parent;
    }

    let canonical = fs.realpathSync.native(cursor);
    for (const part of remainder) {
      canonical = path.join(canonical, part);
    }
    return canonical;
  } catch {
    return null;
  }

function globRegex(glob: string): RegExp {
  let expression = '';
  for (let index = 0; index < glob.length; index += 1) {
    const character = glob[index];
    if (character === '*' && glob[index + 1] === '*') {
      index += 1;
      if (glob[index + 1] === '/') {
        expression += '(?:.*/)?';
        index += 1;
      } else {
        expression += '.*';
      }
    } else if (character === '*') {
      expression += '[^/]*';
    } else if (character === '?') {
      expression += '[^/]';
    } else {
      expression += /[\\^$.*+?()[\]{}|]/.test(character)
        ? `\\${character}`
        : character;
    }
  }
  return new RegExp(`^${expression}$`, 'i');
}

      watcher,
      dataStore,
      parser,
      [markdownProvider, attachmentProvider],
      defaultExtension,
      embeddingProvider
    );

    // Load the features
    const featuresPromises = features.map(feature =>
      feature(context, foamPromise)
    );

    const foam = await foamPromise;
    // Register for disposal right away: a feature failing to load further
    // down must not leave the workspace/graph/watchers alive and untracked
    context.subscriptions.push(foam);
    profiler.stop();
    eventLoopMonitor.stop();

    const resources = foam.workspace.list();
    const noteCount = resources.filter(r => r.type === 'note').length;
    const attachmentCount = resources.filter(
      r => r.type === 'image' || r.type === 'attachment'
    ).length;
    Logger.info(`Loaded ${resources.length} resources`);
    Logger.info(
      profiler.formatReport({
        resources: `${resources.length} (${noteCount} notes, ${attachmentCount} attachments) from ${workspace.workspaceFolders.length} folder(s)`,
        'event loop': eventLoopMonitor.format(),
        memory: formatMemoryUsage(),
      })
    );

    // One broken feature must degrade that feature only, not abort the whole
    // activation (which would also skip extendMarkdownIt for every feature)
    const settled = await Promise.allSettled(featuresPromises);
    settled.forEach((result, i) => {
      if (result.status === 'rejected') {
        Logger.error(
          `Feature #${i} (${
            features[i].name || 'anonymous'
          }) failed to activate`,
          result.reason
        );
      }
    });
    const feats = settled
      .filter(
        (r): r is PromiseFulfilledResult<FoamFeatureResult | void> =>
          r.status === 'fulfilled'
      )
      .map(r => r.value)
      .filter((r): r is FoamFeatureResult => r != null);

    context.subscriptions.push(
      commands.registerCommand('pa-hub.clear-cache', () =>
        parserCache.clear()
      ),
      workspace.onDidChangeConfiguration(e => {
        if (
          [
            'pa-hub.files.ignore',
            'pa-hub.files.exclude',
            'pa-hub.files.include',
            'pa-hub.files.attachmentExtensions',
            'pa-hub.files.noteExtensions',
            'pa-hub.files.defaultNoteExtension',
          ].some(setting => e.affectsConfiguration(setting))
        ) {
          window.showInformationMessage(
            'Foam: Reload the window to use the updated settings'
          );
        }
      })
    );

    return {
      extendMarkdownIt: (md: markdownit) => {
        return feats.reduce((acc: markdownit, r) => {
          return r.extendMarkdownIt ? r.extendMarkdownIt(acc) : acc;
        }, md);
      },
      foam,
    };
  } catch (e) {
    Logger.error('An error occurred while bootstrapping Foam', e);
    window.showErrorMessage(
      `An error occurred while bootstrapping Foam. ${e.stack}`
    );
  }
}

export function handleActiveResourceChange(
  webviewHost: GraphWebviewHost | undefined,
  foam: Foam,
  uri: URI | undefined,
  idToUri: Map<NodeId, URI> = new Map()
) {
  if (webviewHost && uri && uri.scheme !== 'untitled') {
    const note = foam.workspace.find(uri);
    if (isSome(note)) {
      let nodeId: NodeId | undefined;
      for (const [id, resourceUri] of idToUri) {
        if (resourceUri.isEqual(uri)) {
          nodeId = id;
          break;
        }
      }

      if (!nodeId) {
        return;
      }

      webviewHost.webview.postMessage({
        type: 'didSelectNote',
        payload: nodeId,
      });
    }
  }
}

export function parseNodeId(value: unknown): { root: RootKind; path: string } | null {
  if (typeof value !== 'string' || value.length > 1100) {
    return null;
  }

  const separator = value.indexOf(':');
  if (separator <= 0) {
    return null;
  }

  const root = value.slice(0, separator);
  const path = value.slice(separator + 1);
  if (!isRootKind(root) || !isValidNodePath(path)) {
    return null;
  }

  return { root, path };
}
