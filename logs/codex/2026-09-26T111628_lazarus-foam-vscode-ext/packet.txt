```typescript
case 'webviewDidSelectNode': {
  const result = resolveNodeOpen(
    message.payload,
    this.idToUri,
    this.policy,
    uri => isSome(foam.workspace.find(uri))
  );
  if (!result.allowed) {
    break;
  }
  const noteUri = toVsCodeUri(result.uri);
  const navigateToPreview = getFoamVsCodeConfig('graph.navigateToPreview', false);
  const command = getNodeNavigationCommand(result.uri.path, navigateToPreview);
  if (command === 'markdown.showPreview') {
    vscode.commands.executeCommand(command, noteUri);
  } else {
    vscode.commands.executeCommand(command, noteUri, vscode.ViewColumn.One);
  }
  break;
}
```
